\slide{2}{13}{Recap: Bias-variance decomposition}
Green box: $Y = f(X) + \varepsilon$ (``non-random'', ``zero-mean noise'').\par
\textbf{The ``best'' we can do}
\[ \hat f(\mathbf{x}_0) = \mathrm{E}\left[Y \mid X = \mathbf{x}_0\right] \]
{\color{warnc}\textbf{Unknown in practice!!}}\par
{\color{warnc}\textbf{Because the joint distribution of $(X,Y)$ is unknown in practice}}\par
\textbf{What should we do?} \quad {\color{dblue}\textbf{Make assumptions:}}
\[ Y = \beta_0 + X_1\beta_1 + \cdots + X_p\beta_p + \varepsilon, \]
\[ \mathrm{E}[\varepsilon]=0,\quad \varepsilon \textbf{ independent of } (X_1,\dots,X_p) \]
\textbf{Then}
\[ \mathrm{E}[Y\mid X = x_0] = \beta_0 + x_{0_1}\beta_1 + \cdots + x_{0_p}\beta_p \]
\textbf{The \ix{conditional expectation} of $Y$ is {\color{dblue}linear}}

\[ \hat\beta_1 = \frac{\sum_{i=1}^n (x_i-\bar x)(y_i-\bar y)}{\sum_{i=1}^n (x_i-\bar x)^2} \qquad \hat\beta_0 = \bar y - \hat\beta_1\bar x \]

\slide{2}{27}{Bias of the coefficient estimates}
\ixh{bias}\textbf{Least-squares coefficient estimates}
\[ \hat\beta_1 = \frac{\sum_{i=1}^n (x_i-\bar x)(y_i-\bar y)}{\sum_{i=1}^n (x_i-\bar x)^2} \qquad \hat\beta_0 = \bar y - \hat\beta_1\bar x \]
It can be shown that
\[ \mathrm{E}(\hat\beta_1) = \beta_1 \qquad \mathrm{E}(\hat\beta_0) = \beta_0 \]
Recall from the bias definition
\[ \mathrm{Bias}(\hat\beta_1) = \mathrm{E}(\hat\beta_1) - \beta_1 = 0 \qquad \mathrm{Bias}(\hat\beta_0) = \mathrm{E}(\hat\beta_0) - \beta_0 = 0 \]
$\hat\beta_1$ and $\hat\beta_0$ are {\color{warnc}\textbf{\ix{unbiased estimator}}} of $\beta_1$ and $\beta_0$, respectively

\slide{2}{28}{Variance of the coefficient estimates}
\ixh{variance}\textbf{Least-squares coefficient estimates}
\[ \hat\beta_1 = \frac{\sum_{i=1}^n (x_i-\bar x)(y_i-\bar y)}{\sum_{i=1}^n (x_i-\bar x)^2} \qquad \hat\beta_0 = \bar y - \hat\beta_1\bar x \]
It can be shown that\ixh{standard error}
\begin{align*}
\mathrm{Var}(\hat\beta_1) &= \mathrm{SE}(\hat\beta_1)^2 = \frac{\sigma^2}{\sum_{i=1}^n (x_i-\bar x)^2}\\
\mathrm{Var}(\hat\beta_0) &= \mathrm{SE}(\hat\beta_0)^2 = \sigma^2\left[\frac1n + \frac{\bar x^2}{\sum_{i=1}^n (x_i-\bar x)^2}\right]
\end{align*}
where $\sigma^2 = \mathrm{Var}(\varepsilon)$

\slide{2}{29}{Variance of the coefficient estimates}
\textbf{Least-squares coefficient estimates}
\[ \hat\beta_1 = \frac{\sum_{i=1}^n (x_i-\bar x)(y_i-\bar y)}{\sum_{i=1}^n (x_i-\bar x)^2} \qquad \hat\beta_0 = \bar y - \hat\beta_1\bar x \]
In practice, $\sigma^2$ is usually unknown \qquad \textbf{estimate:} $\displaystyle \hat\sigma^2 = \frac{\mathrm{SSR}}{n-2}$\ixh{residual variance estimate}
\begin{align*}
\widehat{\mathrm{SE}}(\hat\beta_1)^2 &= \frac{\hat\sigma^2}{\sum_{i=1}^n (x_i-\bar x)^2}\\
\widehat{\mathrm{SE}}(\hat\beta_0)^2 &= \hat\sigma^2\left[\frac1n + \frac{\bar x^2}{\sum_{i=1}^n (x_i-\bar x)^2}\right]
\end{align*}
\ixh{standard error}

\slide{2}{30}{Accuracy of the model}
\textbf{\ix{residual standard error (RSE)}}
\[ \mathrm{RSE} = \sqrt{\frac{1}{n-2}\mathrm{SSR}} = \sqrt{\frac{1}{n-2}\sum_{i=1}^n (y_i-\hat y_i)^2} \]
\textbf{an absolute measure of lack of fit}\par
\textbf{R squared ($R^2$)}\ixh{R-squared}
\[ \mathrm{R}^2 = 1 - \frac{\mathrm{SSR}}{\mathrm{SST}} = 1 - \frac{\sum_{i=1}^n (y_i-\hat y_i)^2}{\sum_{i=1}^n (y_i-\bar y)^2} \in [0,1] \]
\ixh{total sum of squares (SST)}\textbf{proportion of the variability in $Y$ that can be explained using $X$}\par
Both \textbf{RSE} and $R^2$ favor \textbf{flexible}\ixh{flexibility} methods, which may {\color{warnc}overfit}\ixh{overfitting} the data!!

\slide{2}{35}{Q1: Relationship between response \& predictors}
\ixh{hypothesis test}
\[ H_0: \beta_1 = \beta_2 = \dots = \beta_p = 0 \]
\[ H_a: \textbf{at least one } \beta_j \textbf{ is non-zero} \]
We use the {\color{warnc}\textbf{F-statistics}}\ixh{F-statistic}
\[ F = \frac{(\mathrm{SST}-\mathrm{SSR})/p}{\mathrm{SSR}/(n-p-1)} \qquad \begin{array}{l}\mathrm{SST} = \sum_{i=1}^n (y_i-\bar y)^2\\[2pt] \mathrm{SSR} = \sum_{i=1}^n (y_i-\hat y_i)^2\end{array} \]
\ixh{total sum of squares (SST)}\textbf{When $H_0$ holds,}
\[ \left.\begin{array}{l}(\mathrm{SST}-\mathrm{SSR})/p \approx \sigma^2\\ \mathrm{SSR}/(n-p-1)\approx\sigma^2\end{array}\right\} \longrightarrow\ F \approx 1 \]
A large value of $F$ favors $H_a$

\slide{2}{36}{Q2: Deciding on important variables}
\ixh{variable selection}\ixh{subset selection}
\textbf{\ix{best subset selection}}: compute least squares fit for all possible subsets of predictors, and then choose the ``best''
\begin{itemize}
\item[] ``best'' in terms of some criteria (Mallow's $C_p$\ixh[Cp]{Mallow's $C_p$}, \ix{AIC}, \ix{BIC}, adjusted $R^2$\ixh{adjusted R-squared}, etc.)
\item[] impossible for large $p$: $2^p$ models in total ($\sim$ one billion model when $p=40$)
\end{itemize}
\textbf{\ix{forward selection}}: start with no variables in the model, add variables one-by-one. {\color{dblue}Greedy}\ixh{greedy algorithm} approach.\par
\textbf{\ix{backward selection}}: start with a full model, remove variables one-by-one. Does not work when $p > n$\par
\textbf{\ix{mixed selection}}: combination of forward and backward selection

\slide{2}{38}{Q3: Accuracy of the model}
\textbf{Residual standard error (RSE)}\ixh{residual standard error (RSE)}
\[ \mathrm{RSE} = \sqrt{\frac{1}{n-p-1}\mathrm{SSR}} = \sqrt{\frac{1}{n-p-1}\sum_{i=1}^n (y_i-\hat y_i)^2} \]
\textbf{an absolute measure of lack of fit}\par
\textbf{R squared ($R^2$)}\ixh{R-squared}
\[ \mathrm{R}^2 = 1 - \frac{\mathrm{SSR}}{\mathrm{SST}} = 1 - \frac{\sum_{i=1}^n (y_i-\hat y_i)^2}{\sum_{i=1}^n (y_i-\bar y)^2} \in [0,1] \]
\textbf{proportion of the variability in $Y$ that can be explained using $X$}\par
$R^2$ will {\color{warnc}\textbf{always increase}} as more predictors are added to the model!\par
Both \textbf{RSE} and \textbf{R squared} favor \textbf{flexible}\ixh{flexibility} methods, which may {\color{warnc}\textbf{overfit}}\ixh{overfitting} the data!!

\slide{2}{39}{Adjusted R-squared}
\ixh{adjusted R-squared}
\[ \textit{Adjusted } R^2 = 1 - \frac{(1-R^2)(n-1)}{n-p-1} \]

\slide{2}{44}{Qualitative predictors w/ more than 2 levels}

\subsection{Extensions of the linear model: interactions, polynomial regression}
\ixh{interaction term}\ixh{polynomial regression}

\slide{2}{47}{Multiple regression}
\[ Y = \beta_0 + X_1\beta_1 + \cdots + X_p\beta_p + \varepsilon \]

\slide{2}{48}{Extensions of the linear model}
\[ Y = \beta_0 + X_1\beta_1 + \cdots + X_p\beta_p + \varepsilon \]
\textbf{Relationship between $Y$ and $X_1,\dots,X_p$ is {\color{dblue}additive} and {\color{dblue}linear}}\ixh{additive assumption}\ixh{linearity assumption}\par
\textbf{{\color{dblue}Additive}: the effect of changes in a predictor $X_j$ on $Y$ is independent of the values of all other predictors}\par
\textbf{{\color{dblue}Linear}: the change in the response $Y$ due to one unit change of a predictor $X_j$ is constant, regardless of the value of $X_j$}

\slide{2}{49}{Removing the additive assumption}
\ixh{additive assumption}
\textbf{One way of removing the additive assumption is to include {\color{warnc}interaction} term}\ixh{interaction term}\par
\textbf{from} \quad $Y = \beta_0 + \beta_1X_1 + \beta_2X_2 + \varepsilon$\par
\quad One-unit increase in $X_1$ results in $\beta_1$-unit increase in $Y$ (holding $X_2$ fixed)\par
\textbf{to}
\begin{align*}
Y &= \beta_0 + \beta_1X_1 + \beta_2X_2 + \beta_3\,\boxed{X_1X_2} + \varepsilon\\
 &= \beta_0 + \boxed{(\beta_1 + \beta_3X_2)}\,X_1 + X_2\beta_2 + \varepsilon
\end{align*}
(the blue box $X_1X_2$ is labelled ``{\color{dblue}interaction} between $X_1$ and $X_2$''; the red box marks the slope of $X_1$)\par
One-unit increase in $X_1$ results in $\beta_1 + \beta_3X_2$-unit increase in $Y$\par
the effect of $X_1$ on $Y$ is {\color{warnc}no longer constant}: adjusting $X_2$ will change the impact of $X_1$ on $Y$
