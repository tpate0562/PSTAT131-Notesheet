\slide{3}{10}{A simple logistic model}
\ixh{simple logistic regression}
\textbf{To model $\Pr\big(\text{default}=\mathrm{TRUE}\mid\text{balance}\big)$}
\[
\Pr(Y=1\mid X)=\boxed{\frac{e^{\beta_0+\beta_1X}}{1+e^{\beta_0+\beta_1X}}}\ \ \textcolor{slidec}{\text{\textbf{Logistic function}}}
\qquad \textcolor{dblue}{\in[0,1]}
\]
\ixh{logistic function}
\textbf{which is equivalent to\dots}
\[
\textcolor{slidec}{\text{\textbf{Odds}}}\ \ \boxed{\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}}=e^{\beta_0+\beta_1X}
\qquad \textcolor{dblue}{\in[0,+\infty)}
\]
\ixh{odds}
\textbf{Rolling a 6-sided dice: \ix{odds} of rolling a 3 is 1/5}
\fig{at the left, a dice picture drawn as a fraction: one die (showing six pips) above a fraction bar, and five dice
(showing 5, 4, 3, 2, 1 pips) below it --- one favourable outcome over the five unfavourable ones, i.e.\ odds
$=1/5$ (whereas the probability is $1/6$).}

%% ---------------------------------------------------------------- S3.11
\slide{3}{11}{A simple logistic model}
\[
\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}=e^{\beta_0+\beta_1X}
\]
\textbf{taking logarithm of both sides\dots}
\[
\boxed{\log\left(\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}\right)}=\underline{\beta_0+\beta_1X}
\]
\textcolor{slidec}{\textbf{Logit, or log-odds}}\ixh{logit}\ixh{log odds} \ $\textcolor{dblue}{\in(-\infty,+\infty)}$
\hfill \textcolor{dblue}{\textbf{logit is linear in $X$}}

%% ---------------------------------------------------------------- S3.12
\slide{3}{12}{Interpretation of coefficient}
\ixh{coefficient interpretation}
\[
\log\left(\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}\right)=\beta_0+\beta_1X
\]
\textbf{$\beta_1$: the (\textcolor{dblue}{additive}) change of \ix{logit} by increasing $X$ by 1-unit}\ixh{slope}
\[
\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}=e^{\beta_0+\beta_1X}
\]
\textbf{increasing $X$ by 1-unit means \textcolor{dblue}{multiply} the \ix{odds} by $e^{\beta_1}$}\ixh{odds ratio}

%% ---------------------------------------------------------------- S3.13
\slide{3}{13}{Interpretation of coefficient}
\ixh{coefficient interpretation}
\[
\log\left(\frac{\Pr(Y=1\mid X)}{1-\Pr(Y=1\mid X)}\right)=\beta_0+\beta_1X
\]
\textbf{$\beta_1$: the (\textcolor{dblue}{additive}) change of logit\ixh{logit} by increasing $X$ by 1-unit}

\textbf{$\beta_1$: is \textcolor{slidec}{NOT} the change in $\Pr(Y=1\mid X)$ associated with a 1-unit increase in $X$}
\begin{itemize}
\item[] If $\beta_1>0$ then $X\nearrow\ \Longrightarrow\ \Pr(Y=1\mid X)\nearrow$
\item[] If $\beta_1<0$ then $X\nearrow\ \Longrightarrow\ \Pr(Y=1\mid X)\searrow$
\end{itemize}
\textbf{The \textcolor{slidec}{rate of change} in $\Pr(Y=1\mid X)$ per unit change in $X$ depend on the \textcolor{slidec}{current value} of $X$} \quad \textcolor{dblue}{\textbf{not linear!}}\ixh{nonlinear relationship}
\fig{the ``Logistic Regression'' plot from S3.5: \emph{Probability of Default} (0--1, dashed lines at 0 and 1,
orange rug ticks of the observed 0/1 outcomes) vs.\ \emph{Balance} (0--2500), with the blue S-shaped fitted curve
(flat near 0 until $\approx1000$, steepest around 1800--2000, flattening near 1 by 2600). Written over the plot:
$\Pr(Y=1\mid X)=\dfrac{e^{\beta_0+\beta_1X}}{1+e^{\beta_0+\beta_1X}}$. The slope of the curve is different at
different balances, which is why $\beta_1$ is not a constant change in probability.}

\slide{3}{24}{Multiple logistic regression}
\ixh{multiple logistic regression}
\[
\Pr(Y=1\mid X_1,\ldots,X_p)=\frac{e^{\beta_0+\beta_1X_1+\ldots+\beta_pX_p}}{1+e^{\beta_0+\beta_1X_1+\ldots+\beta_pX_p}}
\ \ \textcolor{dblue}{\in[0,1]}
\]
\ixh{logistic function}
\textbf{which is equivalent to\dots}
\[
\frac{\Pr(Y=1\mid X_1,\ldots,X_p)}{1-\Pr(Y=1\mid X_1,\ldots,X_p)}=e^{\beta_0+\beta_1X_1+\ldots+\beta_pX_p}
\ \ \textcolor{dblue}{\in[0,+\infty)}
\]
\ixh{odds}
\textbf{taking logarithm of both sides\dots}
\begin{align*}
&\log\left(\frac{\Pr(Y=1\mid X_1,\ldots,X_p)}{1-\Pr(Y=1\mid X_1,\ldots,X_p)}\right)\\
&\qquad=\beta_0+\beta_1X_1+\ldots+\beta_pX_p\ \ \textcolor{dblue}{\in(-\infty,+\infty)}
\end{align*}
\ixh{log odds}\ixh{logit}

\slide{3}{25}{Interpretation of coefficient}
\ixh{coefficient interpretation}\ixh{multiple logistic regression}
\[
\log\left(\frac{\Pr(Y=1\mid X_1,\ldots,X_p)}{1-\Pr(Y=1\mid X_1,\ldots,X_p)}\right)=\beta_0+\beta_1X_1+\ldots+\beta_pX_p
\]
\textbf{$\beta_j$: the (\textcolor{dblue}{additive}) change of \ix{logit} by increasing $X_j$ by 1-unit,
\textcolor{dblue}{conditional on the other predictors}}\ixh{holding other predictors fixed}

\begin{tabular}{@{}p{0.30\linewidth}@{\ \ }p{0.66\linewidth}@{}}
\raggedright\textbf{For any $j$, conditional on the other predictors} $X_1,\ldots,X_{j-1},X_j\ldots,X_p$
&
If $\beta_j>0$ then $X_j\nearrow\ \Longrightarrow\ \Pr(Y=1\mid X_1,\ldots,X_p)\nearrow$\par
If $\beta_j<0$ then $X_j\nearrow\ \Longrightarrow\ \Pr(Y=1\mid X_1,\ldots,X_p)\searrow$
\end{tabular}

\ixh{confounding}
\textbf{Pay extra attention when performing regressions involving only a single predictor when other predictors may also be relevant}\ixh{omitted variable}

\textcolor{dblue}{\textbf{Confounding}}\ixh{confounding} \qquad \textcolor{dblue}{\textbf{Simpson's paradox}}\ixh{Simpson's paradox}
\fig{a cartoon scatter plot built from Simpsons characters' heads (black axes with arrows, no labels). The members of
the family form separate groups, each lying along its own \emph{downward}-sloping band: Homer heads and baby Maggie at
the lower left, a column of Bart heads running down from upper-left to lower-right in the middle, a column of Lisa
heads running down to the right of it, and two Marge heads (tall blue hair) at the upper right. A thick red straight
line drawn through all of them rises steeply from lower-left to upper-right: the pooled trend across groups is
positive although the trend \emph{within} each person/group is negative --- the group variable is a confounder
(Simpson's paradox).}

\slide{3}{36}{Many definitions and synonyms}
\ixh{confusion matrix}
A large table (Wikipedia ``Receiver operating characteristic''): \url{https://en.wikipedia.org/wiki/Receiver_operating_characteristic}

\textbf{Core $2\times2$ block} (columns = \textbf{True condition}: Condition positive / Condition negative;
rows = \textbf{Predicted condition}: Predicted condition positive / negative; top-left corner cell: Total population):

\begin{tabular}{@{}l l l@{}}
\toprule
 & Condition positive & Condition negative\\
\midrule
Pred.\ cond.\ positive & \textbf{True positive} & \textbf{False positive} (Type I error)\\
Pred.\ cond.\ negative & \textbf{False negative} (Type II error) & \textbf{True negative}\\
\bottomrule
\end{tabular}
\ixh{true positive}\ixh{false positive}\ixh{false negative}\ixh{true negative}\ixh{Type I error}\ixh{Type II error}

\textbf{Column ratios} (below the block):
\begin{itemize}
\item \textbf{True positive rate (TPR), Recall, Sensitivity, probability of detection}
$=\dfrac{\Sigma\,\text{True positive}}{\Sigma\,\text{Condition positive}}$\ixh{TPR}\ixh{recall}\ixh{sensitivity}
\item \textbf{False positive rate (FPR), Fall-out, probability of false alarm}
$=\dfrac{\Sigma\,\text{False positive}}{\Sigma\,\text{Condition negative}}$\ixh{FPR}\ixh{fall-out}
\item \textbf{False negative rate (FNR), Miss rate}
$=\dfrac{\Sigma\,\text{False negative}}{\Sigma\,\text{Condition positive}}$\ixh{false negative rate (FNR)}
\item \textbf{True negative rate (TNR), Specificity (SPC)}
$=\dfrac{\Sigma\,\text{True negative}}{\Sigma\,\text{Condition negative}}$\ixh{specificity}
\end{itemize}
\textbf{Row ratios} (right of the block):
\begin{itemize}
\item \textbf{Prevalence} $=\dfrac{\Sigma\,\text{Condition positive}}{\Sigma\,\text{Total population}}$\ixh{prevalence}
\item \textbf{Accuracy (ACC)} $=\dfrac{\Sigma\,\text{True positive}+\Sigma\,\text{True negative}}{\Sigma\,\text{Total population}}$\ixh{accuracy}
\item \textbf{Positive predictive value (PPV), Precision} $=\dfrac{\Sigma\,\text{True positive}}{\Sigma\,\text{Predicted condition positive}}$\ixh{precision}\ixh{positive predictive value (PPV)}
\item \textbf{False discovery rate (FDR)} $=\dfrac{\Sigma\,\text{False positive}}{\Sigma\,\text{Predicted condition positive}}$\ixh{false discovery rate (FDR)}
\item \textbf{False omission rate (FOR)} $=\dfrac{\Sigma\,\text{False negative}}{\Sigma\,\text{Predicted condition negative}}$\ixh{false omission rate (FOR)}
\item \textbf{Negative predictive value (NPV)} $=\dfrac{\Sigma\,\text{True negative}}{\Sigma\,\text{Predicted condition negative}}$\ixh{negative predictive value (NPV)}
\end{itemize}
\textbf{Ratios of ratios} (bottom right):
\begin{itemize}
\item \textbf{Positive likelihood ratio (LR+)} $=\dfrac{\text{TPR}}{\text{FPR}}$; \quad
\textbf{Negative likelihood ratio (LR$-$)} $=\dfrac{\text{FNR}}{\text{TNR}}$\ixh{likelihood ratio}
\item \textbf{Diagnostic odds ratio (DOR)} $=\dfrac{\text{LR+}}{\text{LR}-}$\ixh{diagnostic odds ratio}
\item \textbf{F$_1$ score} $=\dfrac{2}{\frac{1}{\text{recall}}+\frac{1}{\text{precision}}}$\ixh{F1 score}
\end{itemize}

cells pink). The TPR cell's denominator ``$\Sigma$ Condition positive'' is highlighted in blue. A small grey legend
thumbnail at the bottom-left repeats the layout with abbreviations --- row 1: pop., C+, C$-$, prev., ACC; row 2: Pc+,
T+, F+, PPV, FDR; row 3: Pc$-$, F$-$, T$-$, FOR (printed ``POR''), NPV; row 4: TPR, FPR, LR+, DOR; row 5: FNR, TNR,
LR$-$ --- with dotted/dashed connector lines and ``+'' / ``$\div$'' signs showing which cells are summed or divided
to form each ratio.}
