%% ---------------------------------------------------------------- S3.36
\slide{3}{36}{Many definitions and synonyms}
\ixh{confusion matrix}
(Wikipedia ``Receiver operating characteristic'' table.) Row ratios:
\textbf{FDR}\ixh{false discovery rate (FDR)}{} $=\Sigma\text{FP}/\Sigma\text{Pred.\ pos.}$;
\textbf{FOR}\ixh{false omission rate (FOR)}{} $=\Sigma\text{FN}/\Sigma\text{Pred.\ neg.}$
Ratios of ratios: \textbf{LR+} $=\text{TPR}/\text{FPR}$, \textbf{LR$-$} $=\text{FNR}/\text{TNR}$\ixh{likelihood ratio};
\textbf{DOR} $=\text{LR+}/\text{LR}-$\ixh{diagnostic odds ratio}.
