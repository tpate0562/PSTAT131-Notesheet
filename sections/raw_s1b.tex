\slide{1}{70}{Bias-variance decomposition}
\ixh{bias-variance decomposition}
\begin{align*}
&\underbrace{E\Big[\big(\tilde{y}_0-\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}})\big)^2\Big]}_{\text{\bfseries expected test MSE}}\\
&\;=\underbrace{\mathit{Var}\big(\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}})\big)}_{\text{\bfseries variance on test data}}
+\underbrace{\big(\mathit{Bias}(\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}}))\big)^2}_{\text{\bfseries bias on test data}}
+\underbrace{\mathit{Var}(\varepsilon)}_{\text{\bfseries irreducible error}}
\end{align*}\ixh{expected test MSE}\ixh{variance}\ixh{bias}\ixh{irreducible error}
\begin{itemize}
\item $\hat{f}$ is learned from a training set\ixh{training data}
\item $(\tilde{\boldsymbol{x}}_{\mathbf{0}},\tilde{y}_0)$ is a test observation, independent of the training set, randomly sampled from the model\ixh{testing data}
\end{itemize}
\[ Y=\underbrace{f(X)}_{\text{\bfseries non-random}}+\underbrace{\varepsilon}_{\text{\bfseries zero-mean noise}} \]\ixh{error term}\ixh{noise}

&E\Big[\big(\tilde{y}_0-\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}})\big)^2\Big]=\mathit{Var}\big(\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}})\big)+\big(\mathit{Bias}(\hat{f}(\tilde{\boldsymbol{x}}_{\mathbf{0}}))\big)^2+\mathit{Var}(\varepsilon)\\

&{\color{dblue}=\mathrm{E}\Big[\big(\hat{f}(\mathbf{x}_0)-\mathrm{E}\hat{f}(\mathbf{x}_0)\big)^2\Big]+\Big[\mathrm{E}\big[\hat{f}(\mathbf{x}_0)\big]-f(\mathbf{x}_0)\Big]^2}+\mathrm{Var}(\varepsilon)
\end{align*}\ixh{expected test MSE}\ixh{variance}\ixh{bias}\ixh{irreducible error}
\textbf{If we take}
\[ {\color{keyc}\hat{f}(\mathbf{x}_0)=\mathrm{E}\big[Y\,|\,X=\mathbf{x}_0\big]} \]\ixh{conditional expectation}
\textbf{then}
\[ {\color{dblue}\mathrm{E}\Big[\big(\hat{f}(\mathbf{x}_0)-\mathrm{E}\hat{f}(\mathbf{x}_0)\big)^2\Big]=0}\quad\textbf{and}\quad{\color{dblue}\Big[\mathrm{E}\big[\hat{f}(\mathbf{x}_0)\big]-f(\mathbf{x}_0)\Big]^2=0} \]

\slide{1}{86}{Bias-variance tradeoff}
\ixh{bias-variance tradeoff}
\begin{align*}
&E\Big[\big(y_0-\hat{f}(\boldsymbol{x}_{\mathbf{0}})\big)^2\Big]=\mathit{Var}\big(\hat{f}(\boldsymbol{x}_{\mathbf{0}})\big)+\big(\mathit{Bias}(\hat{f}(\boldsymbol{x}_{\mathbf{0}}))\big)^2+\mathit{Var}(\varepsilon)\\
&\textbf{\color{warnc}Expected test MSE}\ =\ \textbf{\color{warnc}Variance}\ +\ \textbf{\color{warnc}Bias$^2$}\ +\ \textbf{\color{warnc}Irreducible error}\\
&=\underbrace{{\color{dblue}\mathrm{E}\Big[\big(\hat{f}(\mathbf{x}_0)-\mathrm{E}\hat{f}(\mathbf{x}_0)\big)^2\Big]+\Big[\mathrm{E}\big[\hat{f}(\mathbf{x}_0)\big]-f(\mathbf{x}_0)\Big]^2}}_{\text{\color{dblue}\bfseries minimized!}}+\mathrm{Var}(\varepsilon)\\
&=\underbrace{\mathrm{Var}(\varepsilon)}_{\text{\color{warnc}\bfseries Irreducible error}}
\end{align*}\ixh{irreducible error}
{\color{figgray}\itshape [note: on the slide ``minimized!'' sits under a horizontal line drawn beneath the variance and bias$^2$ terms; this slide writes $y_0$, $\boldsymbol{x}_{\mathbf{0}}$ without tildes.]}

\begin{itemize}

\slide{1}{96}{[untitled --- the Bayes classifier]}
\ixh{Bayes classifier}
\textbf{In classification setting, the ``best'' we can do}
\[ {\color{keyc}\hat{y}_0=\operatorname{argmax}_j\ \Big\{\underbrace{\mathrm{Prob}\big(Y=j\,|\,X=\mathbf{x}_0\big)}_{\substack{\text{\color{warnc}\bfseries conditional probability of $Y=j$,}\\ \text{\color{warnc}\bfseries given the observed predictor $\mathbf{x}_0$}}}\Big\}} \]\ixh{argmax}\ixh{conditional probability}\ixh{posterior probability}
\textbf{{\color{dblue}\ix{Bayes classifier}}: assigns each observation to the most likely class, given its predictor values}

\textbf{For example, we want to classify species of an animal (dog, cat, bird) given its picture}
\[ \left.\begin{aligned}
\mathrm{Prob}\big(Y=\textbf{dog}\,|\,X=\mathbf{x}_0\big)&=0.5\\
\mathrm{Prob}\big(Y=\textbf{cat}\,|\,X=\mathbf{x}_0\big)&=0.3\\
\mathrm{Prob}\big(Y=\textbf{bird}\,|\,X=\mathbf{x}_0\big)&=0.2
\end{aligned}\right\}\ \longrightarrow\ \hat{y}_0=\textbf{dog} \]
