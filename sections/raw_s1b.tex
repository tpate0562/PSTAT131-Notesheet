\slide{1}{61--62}{Lowest training MSE $\neq$ lowest test MSE}
\fig{Ex.~1 (ISLR 2.9): $Y$ (3--11) vs.\ $X$ (0--100); true $f$ (black) rises $\approx$4.5$\to$9.4 (peak $X\approx70$) then falls to $\approx$6.3. Fits by increasing flexibility: orange \ix{linear regression}; light-blue simple \ix{smoothing spline} ($\approx f$); green complicated spline (chases points).
MSE vs.\ flexibility (log, 2--20): test MSE (red) U-shaped: $\approx$2.12 (linear), min $\approx$1.13 at $\approx$5.5 (simple spline), $\approx$1.53 at $\approx$25 (complicated), up to $\approx$2.4. \subsection{Bias-variance decomposition}
\ixh{bias-variance decomposition}\ixh{bias}\ixh{variance}\ixh{irreducible error}\ixh{test MSE}

\slide{1}{81}{Bias-variance tradeoff (Ex.~1)}
\fig{ISLR 2.12 vs.\ flexibility (log 2--20): test MSE U-shaped ($\approx$2.12, min $\approx$1.13 at $\approx$5.5, $\to$2.43); variance rises $\approx$0.05$\to$1.42 (slowly, then sharply); bias$^2$ $\approx$1.08$\to\approx$0 by 5.5, then flat; dashed 1.0 = $\mathrm{Var}(\varepsilon)$; dotted vertical at min. Test MSE = Var + Bias$^2$ + 1.0 everywhere.}

\slide{1}{83--84}{[Ex.~2 and 3 decompositions]}
\fig{Ex.~2 (nearly linear): bias$^2$ tiny ($\approx$0.05 at 2, then $\approx$0); variance $\approx$0.04$\to$1.33; test MSE min $\approx$1.06 at $\approx$3 $\Rightarrow$ optimum at low flexibility, variance dominates. Ex.~3 (highly nonlinear): MSE 18.8$\to$1.2 between 2 and 6, $\to$2.5 at end; bias$^2$ 17.7$\to$0 by $\approx$8; variance $\approx$0 until rising to $\approx$1.5 at right; min at $\approx$10 $\Rightarrow$ bias dominates, optimum at higher flexibility.}
