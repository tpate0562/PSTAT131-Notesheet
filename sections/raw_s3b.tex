\subsection{Bayes' theorem and the idea of discriminant analysis}
\ixh{Bayes' theorem}

\slide{3}{48}{Motivating LDA}
\ixh{linear discriminant analysis (LDA)}
\textbf{Back in Lecture 3, the \ix{Bayes classifier} is the ``best'' we can do}
\[ {\color{keyc}\operatorname{argmax}_{k=1,\ldots,K}\Bigl\{\Pr\bigl(Y=k\mid X_1,\ldots,X_p\bigr)\Bigr\}} \]\ixh{argmax}\ixh{posterior probability}
\textbf{Logistic regression\ixh{logistic regression} ($K=2$ case)}
\[ \Pr(Y=1\mid X_1,\ldots,X_p)=\frac{e^{\beta_0+\beta_1X_1+\ldots+\beta_pX_p}}{1+e^{\beta_0+\beta_1X_1+\ldots+\beta_pX_p}} \]\ixh{logistic function}
\textbf{{\color{dblue}LDA}: uses Bayes theorem\ixh{Bayes' theorem} \& makes assumptions on $X_1,\ldots,X_p\mid Y$}\ixh{linear discriminant analysis (LDA)}

\slide{3}{49}{Bayes theorem}
\ixh{Bayes' theorem}
{\color{dblue}\textbf{Prior probability}}\ixh{prior probability}\quad $\pi_k=\Pr\bigl(Y=k\bigr)$\ixh[pi k]{$\pi_k$ (prior probability)}

{\color{dblue}\textbf{Density function}}\ixh{density function}\ixh{class-conditional density} \textbf{of} $(X_1,\ldots,X_p)\mid Y=k$
\[ f_k(x_1,\ldots,x_p)=\Pr\Bigl(X_1=x_1,\ldots,X_p=x_p\mid Y=k\Bigr) \]\ixh[f k]{$f_k(x)$ (class density)}
{\color{dblue}\textbf{Bayes theorem:}}\ixh{Bayes' theorem}\ixh{posterior probability}
\begin{align*}
&\Pr\Bigl(Y=k\mid X_1=x_1,\ldots,X_p=x_p\Bigr)\\
&\quad=\frac{\Pr\bigl(Y=k,X_1=x_1,\ldots,X_p=x_p\bigr)}{\Pr\bigl(X_1=x_1,\ldots,X_p=x_p\bigr)}\\
&\quad=\frac{\Pr\bigl(Y=k,X_1=x_1,\ldots,X_p=x_p\bigr)}{\sum_{j=1}^K\Pr\bigl(Y=j,X_1=x_1,\ldots,X_p=x_p\bigr)}\\
&\quad=\frac{\pi_kf_k(x_1,\ldots,x_p)}{\sum_{j=1}^K\pi_jf_j(x_1,\ldots,x_p)}
\end{align*}

\hfill{\color{warnc}\textbf{different mean, same variance}}\ixh{common variance assumption}
\fig{``$Y=k$'', ``$\mu_k$'' and ``$\sigma^2$'' are each boxed in orange-red; the annotation ``different mean, same variance'' sits below them.}

\slide{3}{78}{LDA: $p>1$ case}
\textbf{We have multiple \ix{predictor}s $X_1,\ldots,X_p$, which we assume}
\[ (X_1,\ldots,X_p)\mid\boxed{Y=k}\sim N(\boxed{\mu_k},\boxed{\Sigma}) \]\ixh{multivariate normal distribution}\ixh[mu k]{$\mu_k$ (class mean)}\ixh[Sigma]{$\Sigma$ (covariance matrix)}
\begin{center}{\color{warnc}\textbf{different mean, same covariance}}\end{center}\ixh{covariance matrix}\ixh{common covariance assumption}
\[ f_k(\mathbf x)=\frac{1}{(2\pi)^{p/2}|\Sigma|^{1/2}}\exp\left(-\frac12(\mathbf x-\mu_k)^T\Sigma^{-1}(\mathbf x-\mu_k)\right) \]\ixh{density function}\ixh[f k]{$f_k(x)$ (class density)}
with a lot of algebra\ldots

\textbf{The {\color{dblue}Bayes} classifier assigns the observation $\mathbf x$ to class for which}\ixh{Bayes classifier}
\[ \delta_k(\mathbf x)=\mathbf x^T\Sigma^{-1}\mu_k-\frac12\mu_k^T\Sigma^{-1}\mu_k+\log\pi_k \]\ixh{discriminant function}\ixh[delta k]{$\delta_k(x)$ (discriminant function)}
\textbf{is {\color{dblue}largest}}
\fig{``$Y=k$'', ``$\mu_k$'' and ``$\Sigma$'' are boxed in orange-red above the annotation ``different mean, same covariance''.}

\slide{3}{79}{LDA classifier: $p>1$ case}
\ixh{linear discriminant analysis (LDA)}
\textbf{The {\color{dblue}Bayes} classifier assigns the observation $\mathbf x$ to class for which}\ixh{Bayes classifier}
\[ \delta_k(\mathbf x)=\mathbf x^T\Sigma^{-1}\mu_k-\frac12\mu_k^T\Sigma^{-1}\mu_k+\log\pi_k \]\ixh{discriminant function}\ixh[delta k]{$\delta_k(x)$ (discriminant function)}
\textbf{is {\color{dblue}largest}}

\textbf{The {\color{warnc}LDA} classifier assigns the observation $\mathbf x$ to class for which}\ixh{linear discriminant analysis (LDA)}

\subsection{Quadratic discriminant analysis (QDA)}
\ixh{quadratic discriminant analysis (QDA)}

\slide{3}{86}{Quadratic Discriminant Analysis (QDA)}
\ixh{quadratic discriminant analysis (QDA)}
\textbf{Key in Discriminant Analysis}\ixh{discriminant analysis}
\begin{center}$(X_1,\ldots,X_p)\mid Y$ \textbf{is normally distributed}\end{center}\ixh{multivariate normal distribution}
\textbf{In LDA, we assume}\ixh{linear discriminant analysis (LDA)}
\[ (X_1,\ldots,X_p)\mid\boxed{Y=k}\sim N(\boxed{\mu_k},\boxed{\Sigma}) \]\ixh[Sigma]{$\Sigma$ (covariance matrix)}
\begin{center}\textbf{{\color{warnc}different} mean{\color{warnc}, same} covariance}\end{center}\ixh{common covariance assumption}\ixh{covariance matrix}
\textbf{In QDA, we assume}\ixh{quadratic discriminant analysis (QDA)}
\[ (X_1,\ldots,X_p)\mid\boxed{Y=k}\sim N(\boxed{\mu_k},\boxed{\Sigma_k}) \]\ixh[Sigma k]{$\Sigma_k$ (class covariance matrix)}
\begin{center}\textbf{{\color{warnc}different} mean{\color{warnc}, different} covariance}\end{center}\ixh{covariance matrix}
\fig{In both distributions ``$Y=k$'', ``$\mu_k$'' and ``$\Sigma$'' / ``$\Sigma_k$'' are boxed in orange-red.}

\slide{3}{87}{QDA classifier}
\ixh{quadratic discriminant analysis (QDA)}
\textbf{In QDA\ldots}

\textbf{The {\color{dblue}Bayes} classifier assigns the observation $\mathbf x$ to class for which}\ixh{Bayes classifier}
\begin{align*}
\delta_k(\mathbf x)={}&-\frac12\mathbf x^T\Sigma_k^{-1}\mathbf x+\mathbf x^T\Sigma_k^{-1}\mu_k-\frac12\mu_k^T\Sigma_k^{-1}\mu_k\\
&-\frac12\log|\Sigma_k|+\log\pi_k
\end{align*}\ixh{discriminant function}\ixh[delta k]{$\delta_k(x)$ (discriminant function)}
\textbf{is {\color{dblue}largest}}

\textbf{The {\color{warnc}QDA} classifier assigns the observation $\mathbf x$ to class for which}
\begin{align*}
\hat\delta_k(\mathbf x)={}&-\frac12\boxed{\mathbf x^T\hat\Sigma_k^{-1}\mathbf x}+\mathbf x^T\hat\Sigma_k^{-1}\hat\mu_k-\frac12\hat\mu_k^T\hat\Sigma_k^{-1}\hat\mu_k\\
&-\frac12\log|\hat\Sigma_k|+\log\hat\pi_k
\end{align*}\ixh[delta hat k]{$\hat\delta_k(x)$ (estimated discriminant function)}\ixh[Sigma hat k]{$\hat\Sigma_k$ (estimated class covariance matrix)}
\textbf{is {\color{warnc}largest}}
\fig{The term $\mathbf x^T\hat\Sigma_k^{-1}\mathbf x$ is boxed in orange-red and labelled ``{\color{warnc}quadratic in $\mathbf x$}''.}\ixh{quadratic in x}

\slide{3}{90}{Logistic regression vs LDA}
\ixh{logistic regression}\ixh{linear discriminant analysis (LDA)}
\begin{itemize}
\item \textbf{Both produce linear decision boundaries}\ixh{linear decision boundary}\ixh{decision boundary}
\item \textbf{Coefficient estimate procedures are different}\ixh{coefficient}
\item \textbf{LDA assumes {\color{keyc}normal} distribution of predictors given response class}\ixh{normal distribution}
\item \textbf{Logistic regression learns $\Pr(Y=k\mid X)$ directly: {\color{dblue}\ix{discriminative learning}}}\ixh{posterior probability}
\item \textbf{LDA learns $\Pr(Y=k\mid X)$ by modeling $\Pr(Y=k,X)$: {\color{warnc}\ix{generative learning}}}\ixh{joint distribution}
\end{itemize}
